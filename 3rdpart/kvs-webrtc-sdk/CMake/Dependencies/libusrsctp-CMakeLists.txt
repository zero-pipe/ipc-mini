cmake_minimum_required(VERSION 3.6.3)
project(libusrsctp-download NONE)
if(NOT KVS_THIRDPARTY_ROOT)
  message(FATAL_ERROR "KVS_THIRDPARTY_ROOT is required (sibling 3rdpart/ directory)")
endif()
include(ExternalProject)
message(STATUS "Building usrsctp from ${KVS_THIRDPARTY_ROOT}/usrsctp")
ExternalProject_Add(project_libusrsctp
    SOURCE_DIR        ${KVS_THIRDPARTY_ROOT}/usrsctp
    BINARY_DIR        ${CMAKE_CURRENT_BINARY_DIR}/build/src
    DOWNLOAD_COMMAND  ""
    UPDATE_COMMAND    ""
    CMAKE_ARGS        -DCMAKE_INSTALL_PREFIX=${OPEN_SRC_INSTALL_PREFIX}
                      -DCMAKE_BUILD_TYPE=${CMAKE_BUILD_TYPE}
                      "-DCMAKE_C_FLAGS=${CMAKE_C_FLAGS} -fPIC"
                      -Dsctp_werror=0
                      -Dsctp_build_programs=0
                      -DCMAKE_SYSTEM_NAME=${CMAKE_SYSTEM_NAME}
                      -DCMAKE_C_COMPILER=${CMAKE_C_COMPILER}
                      -DCMAKE_TRY_COMPILE_TARGET_TYPE=STATIC_LIBRARY
    BUILD_ALWAYS      TRUE
    TEST_COMMAND      ""
)
